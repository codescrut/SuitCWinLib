        private static string  AddToRequestString(String s, string parameter)
        {
            if (parameter == "") return s;
            if (s != "") s += " ";
            s += parameter;
            return s;

        }
        private static string RequestObjectToString(Request requestObject)
        {
            string s = "";
            if (requestObject.West != "")
            {
                s = AddToRequestString(s, "-W" + requestObject.West);
            }
            if (requestObject.East != "")
            {
                s = AddToRequestString(s, "-E" + requestObject.East);
            }
            if (requestObject.MinCardsWest > 0 && requestObject.MinCardsWest <= 13)
            {
                s = AddToRequestString(s, "-wm" + requestObject.MinCardsWest.ToString());
            }
            if (requestObject.MinCardsEast > 0 && requestObject.MinCardsEast <= 13)
            {
                s = AddToRequestString(s, "-em" + requestObject.MinCardsEast.ToString());
            }
            if (requestObject.VacantPlacesWest >= 0 && requestObject.VacantPlacesWest < 13)
            {
                s = AddToRequestString(s, "-wv" + requestObject.VacantPlacesWest.ToString());
            }
            if (requestObject.VacantPlacesEast >= 0 && requestObject.VacantPlacesEast < 13)
            {
                s = AddToRequestString(s, "-ev" + requestObject.VacantPlacesEast.ToString());
            }
            if (requestObject.NoOfRuffsNorth > 0 && requestObject.NoOfRuffsNorth <= 13)
            {
                s = AddToRequestString(s, "-Rn" + requestObject.NoOfRuffsNorth.ToString());
            }
            if (requestObject.NoOfRuffsSouth > 0 && requestObject.NoOfRuffsSouth <= 13)
            {
                s = AddToRequestString(s, "-Rs" + requestObject.NoOfRuffsSouth.ToString());
            }
            if (requestObject.NoOfRuffsWest > 0 && requestObject.NoOfRuffsWest <= 13)
            {
                s = AddToRequestString(s, "-Rw" + requestObject.NoOfRuffsWest.ToString());
            }
            if (requestObject.NoOfRuffsEast > 0 && requestObject.NoOfRuffsEast <= 13)
            {
                s = AddToRequestString(s, "-Re" + requestObject.NoOfRuffsEast.ToString());
            }
            if (requestObject.LimitNoOfTricks >= 0 && requestObject.LimitNoOfTricks < 13)
            {
                s = AddToRequestString(s, "-t" + requestObject.LimitNoOfTricks.ToString());
            }

            string onLead = "";
            if (requestObject.OnLead == "north")
            {
                onLead = "-Ln";
            }
            else if (requestObject.OnLead == "south")
            {
                onLead = "-Ls";
            }
            if (onLead != "")
            {
                onLead += requestObject.LeadCard;
            }
            if (onLead != "")
            {
                s = AddToRequestString(s, onLead);
            }
            if (requestObject.FreeLeadsNorth < 13)
            {
                s = AddToRequestString(s, "-ln" + requestObject.FreeLeadsNorth.ToString());
            }
            if (requestObject.FreeLeadsSouth < 13)
            {
                s = AddToRequestString(s, "-ls" + requestObject.FreeLeadsSouth.ToString());
            }

            s = AddToRequestString(s, requestObject.North);
            s = AddToRequestString(s, requestObject.South);
            s = AddToRequestString(s, requestObject.WestEast);
            return s;
        }
